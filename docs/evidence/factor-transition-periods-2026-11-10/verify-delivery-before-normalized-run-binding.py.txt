def check(value,label):
    if not value:raise AssertionError(label)

def read(name):return json.loads((OUT/name).read_text())
def git(*args):return subprocess.check_output(['git','-C',str(ROOT),*args])
def log(name):
    p=OUT/name
    return (gzip.decompress(p.read_bytes()) if p.suffix=='.gz' else p.read_bytes()).decode('utf-8')
def optional(name):
    if (OUT/name).exists():return read(name)
    pending.append(name);return None

def statuses(data,label,expected=None):
    if data is None:return
    jobs=data['jobs'];check(expected is None or len(jobs)==expected,label+' job count')
    for j in jobs:
        check(j.get('head_sha')==SOURCE,label+' source binding')
        check(j['conclusion'] not in ('failure','cancelled','timed_out','action_required'),label+' failed job')
    if any(j['status']!='completed' for j in jobs):pending.append(label+' still running');return
    check(all(j['conclusion']=='success' for j in jobs if 'signiert' not in j['name'].lower()),label+' required success')
